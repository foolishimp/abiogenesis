// Diagnostic-only wrappers; no classifier, readiness acknowledgement or timer reset.
import {mkdirSync as diagMkdir,readFileSync as diagRead,writeFileSync as diagWrite,readdirSync as diagList,lstatSync as diagStat} from 'node:fs';
import {createHash as diagHash} from 'node:crypto';
const diagnosticRoot=__DIAGNOSTIC_ROOT__,diagnosticPrelude=__DIAGNOSTIC_PRELUDE__;
const diagnosticLabels=new Set(['resistant-worker','progress-worker','unadmitted-progress-worker','post-timeout-output','post-exit-output','worker-normal','worker-TERM']);
const diagnosticFixtureNames=new Set(['resistant-worker.mjs','progress-worker.mjs','unadmitted-progress-worker.mjs','post-timeout-worker.mjs','post-exit-worker.mjs','post-exit-descendant.mjs',...['worker','command','http'].flatMap(o=>['normal','TERM'].flatMap(m=>[`${o}-${m}-leader.mjs`,`${o}-${m}-descendant.mjs`]))]);
function diagnosticJson(value){const seen=new Map();function encode(v,path){if(typeof v==='bigint')return {type:'BigInt',decimal:String(v)};if(v===undefined)return {type:'undefined'};if(v===null||typeof v!=='object')return v;if(seen.has(v))return {type:'reference',path:seen.get(v)};seen.set(v,path);if(Array.isArray(v))return v.map((x,i)=>encode(x,`${path}/${i}`));const keys=v instanceof Error?[...new Set(['name','message','stack',...Object.getOwnPropertyNames(v)])]:Object.keys(v);return Object.fromEntries(keys.map(k=>[k,encode(v[k],`${path}/${k}`)]));}return JSON.stringify(encode(value,'$'),null,2)+'\n';}
async function writeFile(file,data,...rest){const role=String(file).split('/').at(-1);if(typeof data==='string'&&diagnosticFixtureNames.has(role)){const original=data;const shebang=data.startsWith('#!')?data.slice(0,data.indexOf('\n')+1):'';let body=data.slice(shebang.length);body=body.replace(/\bwriteFileSync\(/gu,'__diagPublishedWrite(').replace(/\bspawn\(process\.execPath/gu,'__diagSpawn(process.execPath');const prelude=diagnosticPrelude.replaceAll('__DIAGNOSTIC_SIDECAR__',JSON.stringify(join(diagnosticRoot,'sidecars',role+'.jsonl'))).replaceAll('__DIAGNOSTIC_ROLE__',JSON.stringify(role));data=shebang+prelude+'\n'+body;diagWrite(join(diagnosticRoot,'fixture-sources',role+'.original'),original,{flag:'wx'});diagWrite(join(diagnosticRoot,'fixture-sources',role+'.instrumented'),data,{flag:'wx'});}return originalWriteFile(file,data,...rest);}
function diagnosticCapture(label,origin,events,result,error,scratch,extra={}){try{const directory=join(diagnosticRoot,'records',label);diagMkdir(directory);const artifacts={};let total=0;function retain(path,name){const b=diagRead(path);total+=b.length;if(b.length>16*1024*1024||total>64*1024*1024)throw Error('diagnostic retained artifact bound');const dest=join(directory,name);diagMkdir(dirname(dest),{recursive:true});diagWrite(dest,b,{flag:'wx'});artifacts[name]={originalPath:path,path:dest,bytes:b.length,sha256:diagHash('sha256').update(b).digest('hex')};}function walk(path,prefix=''){for(const n of diagList(path)){const p=join(path,n),s=diagStat(p);if(s.isSymbolicLink())throw Error('unexpected scratch symlink');if(s.isDirectory())walk(p,prefix+n+'/');else if(s.isFile())retain(p,'scratch/'+prefix+n);}}walk(scratch);for(const n of diagList(join(diagnosticRoot,'sidecars')))retain(join(diagnosticRoot,'sidecars',n),'sidecars/'+n);diagWrite(join(directory,'record.json'),diagnosticJson({kind:'rc11_causal_observation',label,clock:'parent performance.now; child local clocks separate; no cross-process equality assumed',callAtMonotonicMs:origin,returnedAtMs:performance.now()-origin,events,result,error,artifacts,captureComplete:true,...extra}),{flag:'wx'});}catch(e){try{process.stderr.write('RC11_DIAGNOSTIC_CAPTURE_ERROR '+String(e?.stack??e)+'\n');}catch{}}}
async function runWorkerTransport(input){if(!diagnosticLabels.has(input.label))return originalRunWorkerTransport(input);const origin=performance.now(),events=[],original=input.observer,observer={...original},callbackCounts={},originalCallbackCounts={};for(const name of ['onProcessStarted','onStdoutObserved','onStderrObserved','onTimeoutObserved','onSignalRequested','onProcessFinalized','onSpawnFailed']){observer[name]=function(...args){const entry={name,atMs:performance.now()-origin,args,originalCallbackPresent:typeof original?.[name]==='function'};events.push(entry);callbackCounts[name]=(callbackCounts[name]??0)+1;try{if(entry.originalCallbackPresent){originalCallbackCounts[name]=(originalCallbackCounts[name]??0)+1;entry.returnValue=Reflect.apply(original[name],original,args);}return entry.returnValue;}catch(e){entry.error=e;throw e;}finally{entry.returnAtMs=performance.now()-origin;}};}let result,error,threw=false;try{result=await originalRunWorkerTransport({...input,observer});}catch(e){error=e;threw=true;}diagnosticCapture(input.label,origin,events,result,error,input.cwd,{outcome:threw?'throw':'return',timeoutMs:input.timeoutMs,absoluteTimeoutMs:input.absoluteTimeoutMs,terminationGraceMs:input.terminationGraceMs,callbackCounts,originalCallbackCounts});if(threw)throw error;return result;}
async function diagnosticCommand(owner,task,scratch,index){const origin=performance.now();let result,error,threw=false;try{result=await owner(task,scratch,index);}catch(e){error=e;threw=true;}diagnosticCapture(task.taskRef.split('/').at(-1),origin,[],result,error,scratch,{owner:'command',outcome:threw?'throw':'return'});if(threw)throw error;return result;}
async function diagnosticHttp(owner,task,scratch,request){const origin=performance.now();let result,error,threw=false;try{result=await owner(task,scratch,request);}catch(e){error=e;threw=true;}diagnosticCapture(task.taskRef.split('/').at(-1),origin,[],result,error,scratch,{owner:'http',outcome:threw?'throw':'return'});if(threw)throw error;return result;}
