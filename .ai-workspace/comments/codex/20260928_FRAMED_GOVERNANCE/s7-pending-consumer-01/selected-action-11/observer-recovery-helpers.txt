async function preflightObserverAndRecovery(c,e,opened,resumeFrom,recoverFrom){
 const state=await read(join(recoverFrom,'failure-state.json')),selection=(await read(join(recoverFrom,'causal-diagnosis.json'))).safeReentryRoute.selection;
 assert.equal(selection.recoveryCase,'lock_absent');assert.deepEqual(selection.lastCloseHandoff,c.handoff);
 const bytes=await readFile(state.nativePath),frozen=await readFile(join(recoverFrom,'failed-native-prefix.jsonl')),predecessor=await readFile(join(recoverFrom,'predecessor-328.jsonl')),physical=await stat(state.nativePath);
 assert.deepEqual(bytes,frozen);assert.deepEqual(bytes.subarray(0,predecessor.length),predecessor);assert.equal(digest(bytes),selection.expectedCurrent.digest);assert.equal(bytes.length,selection.expectedCurrent.byteLength);assert.equal(physical.dev,c.handoff.reopenAuthority.device);assert.equal(physical.ino,c.handoff.reopenAuthority.inode);assert.equal(digest(predecessor),c.handoff.prefix.prefixDigest);
 const lock=await stat(selection.lockPath).catch(error=>{if(error.code==='ENOENT')return null;throw error;});assert.equal(lock,null);let dead=false;try{process.kill(selection.ownerPid,0);}catch(error){if(error.code!=='ESRCH')throw error;dead=true;}assert.equal(dead,true);
 const actual=c.eventOwner.validateHistoricalEvents(bytes,c.handoff.prefix.storeIdentity.eventContractDigest);assert.equal(actual.length,330);c.abg.selectValidatedRuntimeEventPrefix(actual);
 const retainedBytes=await readFile(join(resumeFrom,'same-basis/interrupted-prefix.jsonl')),retained=c.eventOwner.validateHistoricalEvents(retainedBytes),root=retained.find(row=>row.kind==='basis_admitted'&&row.payload.basisRef===e.executionBasis.basisRef),codec=await c.load('abg/event_body_encoding');assert.ok(root);
 // One explicitly controlled event adds a repeated physical body after a real
 // pending prefix. This is a codec/observer unit assumption, never installed history.
 const {basisRef:_basisRef,basisDigest:_basisDigest,...body}=root.payload,controlledBody={...body,invocationRef:'invocation://selected-action/observer-preflight-only'},basisDigest=c.product.sha256Canonical(controlledBody),basisRef=`execution-basis://abiogenesis/${basisDigest.slice(7)}`;
 const {eventId:_eventId,admissionOrdinal:_ordinal,payloadDigest:_payloadDigest,eventContractDigest:_profile,...candidate}=root;
 const controlled=c.eventOwner.projectRuntimeEventFromValidatedHistory(retained,{...candidate,basisId:basisRef,parentAggregateId:controlledBody.invocationRef,payload:{basisRef,basisDigest,...controlledBody}}),logical=[...retained,controlled],committed=new Map();
 const encoded=logical.map(row=>{const staged=new Map(),line=codec.encodeEventBody(row,committed,staged);for(const[k,v]of staged)committed.set(k,v);return line;}).join(''),physicalInput=Buffer.from(encoded),physicalRows=encoded.trim().split('\n').map(JSON.parse),references=physicalRows.filter(row=>row.kind==='abg_admitted_body_reference_record');assert.ok(references.length>0);assert.equal(references.at(-1).event.admissionOrdinal,logical.length);assert.equal(references.at(-1).admissionOrdinal,undefined);
 const decoded=c.eventOwner.validateHistoricalEvents(physicalInput,c.handoff.prefix.storeIdentity.eventContractDigest);assert.deepEqual(decoded,logical);c.abg.selectValidatedRuntimeEventPrefix(decoded);
 const args={bytes:physicalInput,eventOwner:c.eventOwner,expectedProfileDigest:c.handoff.prefix.storeIdentity.eventContractDigest,call:{invocation:{invocationRef:e.invocationAdmission.publicRequestInvocationRef}},environment:{...e,inputDigest:c.product.sha256Canonical(e.input)},selectionCount:2,abg:c.abg,scopeOwner:c.scopeOwner,truth:c.truth},observed=observePublicSelection(args);assert.ok(observed);assert.equal(observed.invocationAdmission.invocationAdmissionRef,e.invocationAdmission.invocationAdmissionRef);assert.equal(observed.executionBasis.basisRef,e.executionBasis.basisRef);assert.equal(observed.scope.runId,opened.scope.runId);assert.equal(observed.event.eventId,retained.at(-1).eventId);assert.equal(observePublicSelection({...args,call:{invocation:{invocationRef:'invocation://selected-action/not-submitted'}}}),null);
 const nativeError=new Error('controlled native fsync failure');let nativeFailureObserved=false;assert.throws(()=>afterNativeFsync(()=>{throw nativeError;},0,()=>{nativeFailureObserved=true;},()=>{nativeFailureObserved=true;}),error=>error===nativeError);assert.equal(nativeFailureObserved,false);
 let nativeCalls=0,observationFailure;afterNativeFsync(()=>{nativeCalls++;},0,()=>observePublicSelection({...args,bytes:physicalInput.subarray(0,-1)}),error=>{observationFailure={message:error.message,stack:error.stack};});assert.equal(nativeCalls,1);assert.ok(observationFailure);assert.match(observationFailure.message,/incomplete trailing row/);
 await writeFile(join(proof,'controlled-repeated-body.jsonl'),physicalInput);await save(proof,'observer-preflight.json',{status:'passed',proofClass:'controlled additional basis event and storage encoding after actual216 pending prefix; no Product basis admission or installed-history claim',retainedEventCount:retained.length,controlledEvent:controlled,physicalReferenceCount:references.length,physicalReference:references.at(-1),decodedEventCount:decoded.length,correlation:{invocation:e.invocationAdmission.invocationAdmissionRef,basis:e.executionBasis.basisRef,run:opened.scope.runId,selectedEvent:observed.event.eventId},wrongInvocationRefused:true,nativeErrorPropagatesUnchanged:true,observationFailureSeparated:observationFailure});
 await save(proof,'lock-absent-recovery-selection.json',selection);await save(proof,'recovery-precheck.json',{status:'passed',byteLength:bytes.length,digest:digest(bytes),eventCount:actual.length,device:physical.dev,inode:physical.ino,predecessorDigest:digest(predecessor),formerHostDead:dead,lockAbsent:true,lockPath:selection.lockPath,immutablePackageReference:join(resumeFrom,'archive-installed.json')});return selection;
}
